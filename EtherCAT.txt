'ZMC
'xjw
'手动运动调试
'''''''''
wa 1000		'等待驱动器设备上电完成


again:

RAPIDSTOP(2)
wait idle


slot_scan(0)											'开始扫描


if return then 
	
	?"总线扫描成功","连接设备数："NODE_COUNT(0)
	?
	?"开始映射轴号"
	dim num,axisnum
	num=0
	
	for i=0 to NODE_COUNT(0)-1
		axisnum=NODE_AXIS_COUNT(0,i)
		if axisnum<>0 then
			for j=0 to axisnum-1
			AXIS_ADDRESS(num)=num+1			'映射轴号
			ATYPE(num)=65
			units(num)=16770
			'SDO_WRITE (0,0,$6060,0,2,0)
			DRIVE_PROFILE(num)=0			'设置功能
			disable_group(num)					'每轴单独分组
			num=num+1							'总轴数
			next
		endif
		
	next
	

	
	
	?"轴号映射完成","连接总轴数："num
	?
	delay(100)
	
	
	SLOT_START(0)

	if return then 
	?"总线开启成功"
	?
	
	?"开始清除驱动器错误(根据驱动器数据字典设置)"
	for i=0 to num-1 
		DRIVE_CONTROLWORD(i)=128					'根据驱动器数据字典
		delay(10)
		DRIVE_CONTROLWORD(i)=6
		delay(10)
		DRIVE_CONTROLWORD(i)=15
		delay(10)
	next
	
	?"驱动器错误清除完成"
	delay(100)
	?
	
	?"清除控制器错误"
	datum(0)
	?"控制器错误清除完成"
	?
	
	
	
	delay(100)
				
	?"轴使能准备"
	for i=0 to num-1
		base(i)
		AXIS_ENABLE=1
	next
	wdog=1											'使能总开关
	?"轴使能完成"
	?
	
	else
	?"总线开启失败"
	endif
	
	while 1
		if SCAN_EVENT(in(0))>0 then				'in0口查看每个设备带驱动器个数
			?
			for i=0 to NODE_COUNT(0)-1
			?"设备号："i,"带驱动器个数："NODE_AXIS_COUNT(0,i),"设备状态："NODE_STATUS(0,i) 
			next
		
		elseif SCAN_EVENT(in(1))>0 then			'in1口查看io及ad/da参数
			?
			?
			for i=0 to 	NODE_COUNT(0)-1
			?"设备号："i,"起始IO编号："NODE_IO(0,i),"IN个数："NODE_INFO(0,i,10),"OP个数："NODE_INFO(0,i,11) 
			?,,"起始AD/DA："NODE_AIO(0,i),"AD个数："NODE_INFO(10,i,12),"DA个数："NODE_INFO(10,i,13)
			?
			next
			
		elseif SCAN_EVENT(in(2))>0 then
				
			?
			for i=0 to NODE_COUNT(0)-1
				?*ETHERCAT(i)
				?
			next
				
			
			
		elseif SCAN_EVENT(in(3))>0 then			'查看所有设备状态
			?
			?*ETHERCAT
			

		
		elseif SCAN_EVENT(in(4))>0 then
			
			vmove(1)axis(0)
	
		elseif SCAN_EVENT(in(5))>0 then
			
			vmove(-1)axis(0)
			
		elseif SCAN_EVENT(in(6))>0 then
			
			RAPIDSTOP(2)
			
		elseif SCAN_EVENT(in(7))>0 then
			
			SDO_WRITE (0,0,$6060,0,2,0)
			?11
			
		
		elseif SCAN_EVENT(in(22))>0 then
		
			again
			?"启动总线，等待1s"
			delay(1000)
		
		elseif SCAN_EVENT(in(23))>0 then
		
			SLOT_STOP(0)
			?"停止总线"
			
		endif
		
		
		
	wend
	
else
	?"总线扫描失败"
endif

end
